@app.route('/inventory')
@login_required
def inventory():
    weapons = Weapon.query.filter_by(user_id=current_user.id).all()
    skill_books = SkillBook.query.filter_by(user_id=current_user.id).all()
    return render_template('inventory.html', weapons=weapons, skill_books=skill_books)

@app.route('/shop')
@login_required
def shop():
    available_weapons = [
        {'name': 'Iron Sword', 'tier': 1, 'price': 100},
        {'name': 'Steel Blade', 'tier': 2, 'price': 500},
        {'name': 'Mystic Spear', 'tier': 3, 'price': 2000}
    ]
    available_skills = [
        {'name': 'Fireball', 'tier': 1, 'price': 150},
        {'name': 'Lightning Strike', 'tier': 2, 'price': 600},
        {'name': 'Meteor Storm', 'tier': 3, 'price': 2500}
    ]
    return render_template('shop.html', weapons=available_weapons, skills=available_skills)

@app.route('/shop/buy', methods=['POST'])
@login_required
def buy_item():
    data = request.get_json()
    item_type = data.get('type')  # 'weapon' or 'skill'
    item_name = data.get('name')
    item_tier = data.get('tier')
    item_price = data.get('price')
    
    if current_user.gold < item_price:
        return jsonify({'status': 'error', 'message': 'Not enough gold'}), 400
    
    current_user.gold -= item_price
    
    if item_type == 'weapon':
        new_weapon = Weapon(name=item_name, tier=item_tier, user_id=current_user.id)
        db.session.add(new_weapon)
    elif item_type == 'skill':
        new_skill = SkillBook(name=item_name, tier=item_tier, user_id=current_user.id)
        db.session.add(new_skill)
    
    db.session.commit()
    return jsonify({'status': 'success', 'gold': current_user.gold})

@app.route('/inventory/merge', methods=['POST'])
@login_required
def merge_items():
    data = request.get_json()
    item_type = data.get('type')
    item_ids = data.get('item_ids')  # List of 3 items to merge
    
    if len(item_ids) != 3:
        return jsonify({'status': 'error', 'message': 'Need exactly 3 items to merge'}), 400
    
    if item_type == 'weapon':
        items = Weapon.query.filter(Weapon.id.in_(item_ids), Weapon.user_id == current_user.id).all()
    else:
        items = SkillBook.query.filter(SkillBook.id.in_(item_ids), SkillBook.user_id == current_user.id).all()
    
    if len(items) != 3:
        return jsonify({'status': 'error', 'message': 'Invalid items selected'}), 400
    
    # Check all items are same tier
    if len(set(item.tier for item in items)) != 1:
        return jsonify({'status': 'error', 'message': 'All items must be same tier'}), 400
    
    old_tier = items[0].tier
    new_tier = old_tier + 1
    
    # Delete old items
    for item in items:
        db.session.delete(item)
    
    # Create new merged item
    if item_type == 'weapon':
        new_item = Weapon(
            name=f"{items[0].name} +{new_tier}",
            tier=new_tier,
            user_id=current_user.id
        )
    else:
        new_item = SkillBook(
            name=f"{items[0].name} +{new_tier}",
            tier=new_tier,
            user_id=current_user.id
        )
    
    db.session.add(new_item)
    db.session.commit()
    
    return jsonify({'status': 'success', 'new_tier': new_tier})

        const selected = type === 'weapon' ? selectedWeapons : selectedSkills;

from functools import wraps

def admin_required(f):
    @wraps(f)
    def decorated_function(*args, **kwargs):
        if not current_user.is_authenticated or not current_user.is_admin:
            flash('Admin access required')
            return redirect(url_for('index'))
        return f(*args, **kwargs)
    return decorated_function

@app.route('/admin')
@login_required
@admin_required
def admin_dashboard():
    total_users = User.query.count()
    total_gold = db.session.query(db.func.sum(User.gold)).scalar() or 0
    pending_cashouts = CashoutRequest.query.filter_by(status='pending').all()
    recent_users = User.query.order_by(User.id.desc()).limit(10).all()
    
    return render_template('admin.html',
                         total_users=total_users,
                         total_gold=total_gold,
                         pending_cashouts=pending_cashouts,
                         recent_users=recent_users)

        user.gold = data['gold']

@login_required
@admin_required
def approve_cashout(cashout_id):
    cashout = CashoutRequest.query.get_or_404(cashout_id)
    
    if cashout.status != 'pending':
        return jsonify({'status': 'error', 'message': 'Already processed'}), 400
    
    cashout.status = 'approved'
    cashout.processed_at = datetime.utcnow()
    db.session.commit()
    
    return jsonify({'status': 'success'})

@app.route('/admin/cashout/<int:cashout_id>/reject', methods=['POST'])
@login_required
@admin_required
def reject_cashout(cashout_id):
    cashout = CashoutRequest.query.get_or_404(cashout_id)
    
    if cashout.status != 'pending':
        return jsonify({'status': 'error', 'message': 'Already processed'}), 400
    
    # Refund the user
    user = User.query.get(cashout.user_id)
    user.gold += 30000  # Refund full amount
    
    cashout.status = 'rejected'
    cashout.processed_at = datetime.utcnow()
    db.session.commit()
    
    return jsonify({'status': 'success'})

        if (!confirm('Ban this user?')) return;
        
        const response = await fetch(`/admin/user/${userId}/edit`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ is_banned: true })
        });
        
        const data = await response.json();
        if (data.status === 'success') {
            alert('User banned!');
            document.getElementById(`user-${userId}`).style.opacity = '0.5';
        }
    }
