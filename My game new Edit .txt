"""Combat mechanics for Cultivator Era."""

import random
from config import Config


def calculate_damage(attacker_power, defender_defense):
    """Calculate damage dealt in combat."""
    base_damage = max(1, attacker_power - defender_defense // 2)
    # Add randomness ±20%
    variance = random.uniform(0.8, 1.2)
    return int(base_damage * variance)


def calculate_monster_rewards(monster_level, player_level):
    """Calculate rewards for killing a monster."""
    level_diff = monster_level - player_level
    bonus_multiplier = 1.0
    
    if level_diff > 0:
        # Higher level monster = better rewards
        bonus_multiplier = 1.0 + (level_diff * Config.LEVEL_DIFFERENCE_BONUS)
    
    gold = int(Config.MONSTER_BASE_GOLD * monster_level * bonus_multiplier)
    exp = int(Config.MONSTER_BASE_EXP * monster_level * bonus_multiplier)
    agility = int(Config.MONSTER_BASE_AGILITY * monster_level * bonus_multiplier)
    
    return {
        "gold": gold,
        "exp": exp,
        "agility": agility,
        "bonus_multiplier": bonus_multiplier,
    }


def generate_loot_drop(monster_level):
    """Random loot drop from monsters."""
    drop_chance = min(0.3, 0.1 + (monster_level * 0.02))
    
    if random.random() < drop_chance:
        item_types = ["weapon", "armor", "skill_book"]
        item_type = random.choice(item_types)
        
        # Grade based on monster level
        if monster_level >= 20:
            grades = ["Epic", "Legendary"]
        elif monster_level >= 10:
            grades = ["Rare", "Epic"]
        elif monster_level >= 5:
            grades = ["Common", "Rare"]
        else:
            grades = ["Common"]
        
        grade = random.choice(grades)
        
        # Generate item stats
        grade_multipliers = {
            "Common": 1.0,
            "Rare": 1.5,
            "Epic": 2.5,
            "Legendary": 4.0,
        }
        
        mult = grade_multipliers[grade]
        
        if item_type == "weapon":
            return {
                "type": "weapon",
                "name": f"{grade} {random.choice(['Sword', 'Axe', 'Spear', 'Dagger'])}",
                "grade": grade,
                "attack_bonus": int(5 * monster_level * mult),
                "defense_bonus": 0,
            }
        elif item_type == "armor":
            return {
                "type": "armor",
                "name": f"{grade} {random.choice(['Helmet', 'Chestplate', 'Boots', 'Gloves'])}",
                "grade": grade,
                "attack_bonus": 0,
                "defense_bonus": int(3 * monster_level * mult),
            }
        else:  # skill_book
            return {
                "type": "skill_book",
                "name": f"{grade} Skill: {random.choice(['Fireball', 'Ice Shard', 'Lightning Strike', 'Heal'])}",
                "grade": grade,
                "attack_bonus": int(3 * monster_level * mult),
                "defense_bonus": int(2 * monster_level * mult),
            }
    
    return None


def apply_spiritual_root_bonus(player, damage):
    """Apply spiritual root bonus to damage."""
    root_bonuses = {
        "Fire": 1.15,
        "Water": 1.05,
        "Earth": 1.05,
        "Wind": 1.10,
        "Lightning": 1.20,
        "Ice": 1.12,
        "Dark": 1.18,
        "Light": 1.10,
    }
    
    multiplier = root_bonuses.get(player.spiritual_root, 1.0)
    multiplier += (player.spiritual_root_level - 1) * 0.05
    
    return int(damage * multiplier)


def simulate_combat_turn(player, enemy, is_player_turn=True):
    """Simulate one turn of combat."""
    if is_player_turn:
        damage = calculate_damage(player.attack_power, enemy["defense"])
        damage = apply_spiritual_root_bonus(player, damage)
        enemy["hp"] -= damage
        return {
            "attacker": player.username,
            "defender": enemy["name"],
            "damage": damage,
            "defender_hp": max(0, enemy["hp"]),
        }
    else:
        damage = calculate_damage(enemy["attack_power"], player.defense)
        player.hp -= damage
        return {
            "attacker": enemy["name"],
            "defender": player.username,
            "damage": damage,
            "defender_hp": max(0, player.hp),
        }

"""Dungeon generation and monster spawning."""

import random
from models import db, Monster, Dungeon


def get_or_create_dungeon(level_range_start, level_range_end):
    """Get or create a dungeon for a level range."""
    dungeon = Dungeon.query.filter_by(
        min_level=level_range_start,
        max_level=level_range_end
    ).first()
    
    if not dungeon:
        dungeon = Dungeon(
            name=f"Dungeon Lv.{level_range_start}-{level_range_end}",
            min_level=level_range_start,
            max_level=level_range_end,
            max_floors=10,
            description=f"A dangerous dungeon for cultivators level {level_range_start} to {level_range_end}."
        )
        db.session.add(dungeon)
        db.session.commit()
    
    return dungeon


def spawn_monster(dungeon, floor_number):
    """Spawn a monster appropriate for the dungeon and floor."""
    # Monster level scales with dungeon level and floor
    min_level = dungeon.min_level + (floor_number - 1)
    max_level = min(dungeon.max_level, min_level + 3)
    monster_level = random.randint(min_level, max_level)
    
    monster_names = [
        "Corrupted Wolf", "Shadow Beast", "Stone Golem", "Fire Elemental",
        "Ice Wraith", "Dark Sorcerer", "Ancient Guardian", "Void Demon",
        "Crystal Dragon", "Thunder Phoenix", "Abyssal Horror", "Celestial Warrior"
    ]
    
    name = random.choice(monster_names)
    
    # Stats scale with level
    hp = 50 + (monster_level * 20)
    attack = 5 + (monster_level * 3)
    defense = 3 + (monster_level * 2)
    
    # Rewards
    gold = 50 * monster_level
    exp = 50 * monster_level
    agility = 50 * monster_level
    
    monster = Monster(
        name=f"{name} (Lv.{monster_level})",
        level=monster_level,
        hp=hp,
        attack_power=attack,
        defense=defense,
        gold_reward=gold,
        exp_reward=exp,
        agility_reward=agility,
        dungeon_id=dungeon.id
    )
    
    db.session.add(monster)
    db.session.commit()
    
    return monster


def get_available_dungeons(player_level):
    """Get list of dungeons available to the player."""
    dungeons = []
    
    # Create dungeon ranges: 1-10, 10-20, 20-30, etc.
    for i in range(0, 10):
        start = max(1, i * 10)
        end = (i + 1) * 10
        if player_level >= start - 5:  # Can enter dungeons 5 levels below
            dungeons.append({
                "min_level": start,
                "max_level": end,
                "name": f"Dungeon Lv.{start}-{end}",
                "recommended": start <= player_level <= end,
            })
    
    return dungeons
